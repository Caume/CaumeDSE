HerraduraKEx independent primitive test vectors
================================================

Purpose
-------

These vectors support `testHerraduraIndependent()` in `function_tests.c`. They
exercise upstream `herradura.h` primitives directly, without CaumeDSE SQLite
storage, PBKDF, or `CDSEHKX1` framing.

Upstream reference
------------------

- Repository: `Caume/HerraduraKEx`
- Reviewed commit: `13e5fb0346ca5ec81202dee8bb3302633780ec35`
- Header under test: `herradura.h`
- Review status: CDSE-generated deterministic vectors from the reviewed header;
  not upstream-published vectors.

Generator
---------

Build and run the generator from a HerraduraKEx include directory:

```sh
gcc -x c -I/path/to/HerraduraKEx -o /tmp/cdse_hkx_vectors - <<'EOF'
#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include <herradura.h>

static void print_hex(const char *label, const uint8_t *buf, size_t len)
{
    size_t i;
    printf("%s=", label);
    for (i=0; i<len; i++) printf("%02x", buf[i]);
    printf("\n");
}

static void fill_key(BitArray *key)
{
    int i;
    for (i=0; i<KEYBYTES; i++) key->b[i]=(uint8_t)(0x10+i*3);
}

static void fill_nonce(BitArray *nonce)
{
    int i;
    for (i=0; i<KEYBYTES; i++) nonce->b[i]=(uint8_t)(0xa0-i*5);
}

int main(void)
{
    BitArray key, nonce;
    const uint8_t ad[] = "cdse-independent-hkx-test-aad-v1";
    uint8_t pt[97], ct[sizeof(pt)], tag[KEYBYTES], dec[sizeof(pt)], hash[KEYBYTES];
    size_t i;

    fill_key(&key);
    fill_nonce(&nonce);
    for (i=0; i<sizeof(pt); i++) pt[i]=(uint8_t)((i*7+3)&0xff);

    print_hex("key", key.b, KEYBYTES);
    print_hex("nonce", nonce.b, KEYBYTES);
    print_hex("ad", ad, sizeof(ad)-1);
    print_hex("pt97", pt, sizeof(pt));

    hske_nl_aead_encrypt(&key, &nonce, ad, sizeof(ad)-1, pt, sizeof(pt), ct, tag);
    print_hex("nla1_ct97", ct, sizeof(ct));
    print_hex("nla1_tag97", tag, KEYBYTES);
    printf("nla1_dec=%d\n", hske_nl_aead_decrypt(&key, &nonce, ad, sizeof(ad)-1, ct, sizeof(ct), tag, dec));
    printf("nla1_plain_match=%d\n", memcmp(pt, dec, sizeof(pt))==0);

    hske_nl_v2_duplex_encrypt(&key, &nonce, ad, sizeof(ad)-1, pt, sizeof(pt), ct, tag);
    print_hex("duplex_ct97", ct, sizeof(ct));
    print_hex("duplex_tag97", tag, KEYBYTES);
    printf("duplex_dec=%d\n", hske_nl_v2_duplex_decrypt(&key, &nonce, ad, sizeof(ad)-1, ct, sizeof(ct), tag, dec));
    printf("duplex_plain_match=%d\n", memcmp(pt, dec, sizeof(pt))==0);

    hske_nl_aead_encrypt(&key, &nonce, ad, sizeof(ad)-1, pt, 0, ct, tag);
    print_hex("nla1_tag0", tag, KEYBYTES);
    printf("nla1_dec0=%d\n", hske_nl_aead_decrypt(&key, &nonce, ad, sizeof(ad)-1, ct, 0, tag, dec));

    hske_nl_v2_duplex_encrypt(&key, &nonce, ad, sizeof(ad)-1, pt, 0, ct, tag);
    print_hex("duplex_tag0", tag, KEYBYTES);
    printf("duplex_dec0=%d\n", hske_nl_v2_duplex_decrypt(&key, &nonce, ad, sizeof(ad)-1, ct, 0, tag, dec));

    hfscx_256(pt, sizeof(pt), NULL, hash);
    print_hex("hfscx97", hash, KEYBYTES);
    hfscx_256_ds(0x03, pt, sizeof(pt), NULL, hash);
    print_hex("hfscxds03_97", hash, KEYBYTES);
    return 0;
}
EOF
/tmp/cdse_hkx_vectors
```
